Purpose: The channel-lifecycle pattern is the highest-leverage code in the whole phase (per 03-RESEARCH.md §1: "open the channel exactly once and clean it up"). Getting it wrong leaks WebSockets across StrictMode double-mounts and produces ghost participants. Isolating this into Plan 04 with a stub Player keeps the scope tight. Plans 05/06 then attach hooks/state without rewriting the channel boilerplate.

Output: A working route at `/room/[id]` that, given a valid cookie, renders the broadcast header + a (still-empty-state-aware) Player and opens a Realtime channel. The channel currently has no broadcast handlers — Plan 05 attaches them. Manually visiting the route in dev shows the OnAir/RoomHeader UI with city + count placeholder; the WebSocket connects and is cleanly torn down on navigation.

    import { useEffect, useState } from 'react';
    import type { RealtimeChannel } from '@supabase/supabase-js';

    import { createClient } from '@/lib/supabase/client';
    import { Player } from '@/components/room/Player';
    import { RoomHeader } from '@/components/room/RoomHeader';

    type Props = {
      signup: { id: string; name: string; city: string };
      room: {
        id: string;
        type: string;
        city: string;
        youtubeVideoId: string | null;
      };
    };

    /**
     * RoomClient: opens exactly ONE Realtime channel per mount and cleans it up on unmount.
     * Cleanup pairs unsubscribe() AND removeChannel(channel) — both required to avoid
     * leaks across React 19 StrictMode double-mount and hot-reload (03-RESEARCH.md §1,
     * Pitfall 1).
     *
     * Plan 04 (this plan) wires the channel lifecycle and renders the Header + Player skeleton.
     * Plan 05 will attach: usePresence (host election), useRoomSync (broadcast handlers,
     * heartbeat, drift correction), VideoPickerSheet.
     * Plan 06 will attach: Chat panel and chat broadcast handlers.
     *
     * The channel is owned here. Hooks in Plan 05/06 will receive it as a prop, NOT create
     * their own — one channel per room per client (PATTERNS.md "Pattern 1").
     */
    export default function RoomClient({ signup, room }: Props) {
      const [channel, setChannel] = useState<RealtimeChannel | null>(null);
      const [videoId] = useState<string | null>(room.youtubeVideoId);

      useEffect(() => {
        const supabase = createClient();
        const ch = supabase.channel(`room:${room.id}`, {
          config: {
            presence: { key: signup.id },
            broadcast: { self: false },
          },
        });

        ch.subscribe(async (status) => {
          if (status === 'SUBSCRIBED') {
            // Re-track on EVERY 'SUBSCRIBED' (mobile-bg reconnect — Pitfall 7).
            // Payload locked by C-007 + RESEARCH.md addition: { user_id, name, city, joined_at }.
            await ch.track({
              user_id: signup.id,
              name: signup.name,
              city: signup.city,
              joined_at: Date.now(),
            });
          }
        });

        setChannel(ch);

        return () => {
          ch.unsubscribe();
          supabase.removeChannel(ch);
          setChannel(null);
        };
      }, [room.id, signup.id, signup.name, signup.city]);

      // Plan 05 will derive these from usePresence(channel, signup.id):
      //   const { participants, host, isHost, presenceReady, count } = usePresence(channel, signup.id);
      // For Plan 04 (skeleton), provide stub values so the Header renders:
      const isHost = false;
      const hostName: string | null = null;
      const count = 1; // self-only until Plan 05 wires presence

      return (
        <main className="min-h-screen relative overflow-hidden flex flex-col bg-[color:var(--bg)]">
          <RoomHeader city={room.city} count={count} />

          <section className="flex-1 mx-auto max-w-6xl w-full px-6 py-8 flex flex-col gap-6">
            <div className="rise" style={{ animationDelay: '120ms' }}>
              <Player
                videoId={videoId}
