Wire presence, sync, and host-only video selection on top of the Plan 04 channel skeleton. Two hooks (`usePresence`, `useRoomSync`) own the Realtime broadcast handlers and the heartbeat interval; two view components (`PresenceList`, `VideoPickerSheet`) render the participant strip and the host's curation modal. RoomClient is updated to thread the hooks together: channel from Plan 04 → presence → host election → sync wiring → Player props.

Purpose: This plan closes REQ-ROOM-SYNC and REQ-PRESENCE. The hooks are deliberately small and isolated — every cross-client invariant (host election, drift threshold, cooldown, lookahead) is locked in 03-RESEARCH.md and re-uses the pure functions already shipped in Plan 01. Three pitfalls from RESEARCH.md drive the structure: (1) one channel per room (hooks receive it as a prop, never create it); (2) `suppressNextEvent` ref breaks the seek→onStateChange→broadcast→seek loop; (3) 1-second cooldown prevents the heartbeat-after-correction → re-correction loop.

Output: A working room where two browser tabs see each other in the presence list, the first joiner has the HostBadge and player controls, host pause/play/seek mirrors to the other tab within ~2s, throttling one tab triggers a sync_correct from the host.
</objective>

    'use client';

    import { useEffect, useMemo, useState } from 'react';
    import type { RealtimeChannel } from '@supabase/supabase-js';

    import type { Participant } from '@/lib/room-types';
    import { electHost } from '@/lib/sync-utils';

    /**
     * usePresence: subscribes to channel.on('presence', { event: 'sync' }) and
     * derives the canonical { participants, host, isHost, presenceReady, count }.
     *
     * The channel is owned by RoomClient (one channel per room — PATTERNS.md "Pattern 1").
     * This hook only attaches a listener; no track() call (RoomClient does that in its
     * subscribe callback so it survives reconnects — Pitfall 7).
     *
     * presenceReady gates UI off the first 'sync' event (RESEARCH.md §4): until then,
     * presenceState() may return only the self entry which leads to false-positive host
     * election for the brief window between SUBSCRIBED and the first sync.
     */
    export function usePresence(
      channel: RealtimeChannel | null,
      selfId: string,
    ) {
      const [state, setState] = useState<Record<string, Participant[]>>({});
      const [presenceReady, setPresenceReady] = useState(false);

      useEffect(() => {
        if (!channel) {
          setPresenceReady(false);
          return;
        }
        channel.on('presence', { event: 'sync' }, () => {
          // presenceState is generic; cast to our typed shape (track payload is C-007).
          const next = channel.presenceState<Participant>() as Record<string, Participant[]>;
          setState(next);
          setPresenceReady(true);
        });
        // No cleanup-listener API in supabase-js v2; channel removal in RoomClient
        // tears down all listeners. Just clear local state when channel detaches.
        return () => {
          setState({});
          setPresenceReady(false);
        };
      }, [channel]);

      const participants = useMemo<readonly Participant[]>(
        () => Object.values(state).flat(),
        [state],
      );

      const host = useMemo(() => electHost(participants), [participants]);
      const isHost = host?.user_id === selfId;

      return {
        participants,
        host,
        isHost,
        presenceReady,
        count: participants.length,
      };
    }
    ```

    Notes:
    - The hook imports `electHost` from `lib/sync-utils.ts` rather than re-implementing the sort. Single source of truth → unit-tested.
    - `presenceReady` is the recommended UI gate (RESEARCH.md §4). Without it, the very first render of `<PresenceList>` after `SUBSCRIBED` may show only self.
    - On channel disconnect, state resets to `{}` and `presenceReady` to `false` — keeps stale state from leaking across re-mounts.

    'use client';

    import { useEffect, useRef } from 'react';
    import type { RealtimeChannel } from '@supabase/supabase-js';
    import type { YouTubePlayer } from 'react-youtube';

    import { shouldCorrect } from '@/lib/sync-utils';

    /**
     * useRoomSync: attaches all broadcast handlers for the Phase 3 sync protocol on the
     * given channel. Locked behavior:
     *   - sync_play / sync_pause / sync_seek: every client (including host) seeks the
     *     player to the broadcast timestamp; play/pause as appropriate.
     *   - heartbeat: every client emits every 5s; ONLY the host computes drift via
     *     shouldCorrect() and replies with sync_correct targeted to the drifting user_id.
     *   - sync_correct: client filters by target_user_id; on match, seeks and starts a
     *     1-second cooldown that suppresses the next heartbeat (Pitfall 3).
     *
     * Lookahead +0.3s on sync_correct payload absorbs the round-trip latency between
     * host.getCurrentTime() and client.seekTo() (RESEARCH.md §3).
     *
     * The host's own onStateChange handler in RoomClient/Player uses the returned
     * suppressNextEvent ref to break the host's-own-seekTo → onStateChange → broadcast loop
     * (Pitfall 4).
     */
    export function useRoomSync(
      channel: RealtimeChannel | null,
      selfId: string,
      isHost: boolean,
      playerRef: React.RefObject<YouTubePlayer | null>,
    ) {
      const correctionCooldownUntil = useRef(0);
      const suppressNextEvent = useRef(false);

      // Attach broadcast handlers
      useEffect(() => {
        if (!channel) return;

        channel
          .on('broadcast', { event: 'sync_play' }, ({ payload }) => {
            const player = playerRef.current;
            if (!player) return;
            player.seekTo(payload.timestamp, true);
            player.playVideo();
          })
          .on('broadcast', { event: 'sync_pause' }, ({ payload }) => {
            const player = playerRef.current;
            if (!player) return;
            player.seekTo(payload.timestamp, true);
            player.pauseVideo();
          })
          .on('broadcast', { event: 'sync_seek' }, ({ payload }) => {
            playerRef.current?.seekTo(payload.timestamp, true);
          })
          .on('broadcast', { event: 'sync_correct' }, ({ payload }) => {
            if (payload.target_user_id !== selfId) return;
            playerRef.current?.seekTo(payload.timestamp, true);
            // 1-second cooldown — next heartbeat skipped (Pitfall 3).
            correctionCooldownUntil.current = Date.now() + 1000;
          })
          .on('broadcast', { event: 'heartbeat' }, ({ payload }) => {
            // Only the host computes drift.
            if (!isHost) return;
            const player = playerRef.current;
            if (!player) return;
            const hostTime = player.getCurrentTime();
            if (shouldCorrect(hostTime, payload.currentTime)) {
              channel.send({
                type: 'broadcast',
                event: 'sync_correct',
                payload: {
                  target_user_id: payload.user_id,
                  // +0.3s lookahead absorbs round-trip latency (RESEARCH.md §3).
                  timestamp: hostTime + 0.3,
                },
              });
            }
          });
      }, [channel, isHost, selfId, playerRef]);

      // 5-second heartbeat from this client (every client, host included)
      useEffect(() => {
        if (!channel) return;
        const id = setInterval(() => {
          // Skip during correction cooldown (Pitfall 3).
          if (Date.now() < correctionCooldownUntil.current) return;
          const player = playerRef.current;
          if (!player) return;
          const ct = player.getCurrentTime();
          channel.send({
            type: 'broadcast',
            event: 'heartbeat',
            payload: { user_id: selfId, currentTime: ct, sentAt: Date.now() },
          });
        }, 5000);
        return () => clearInterval(id);
      }, [channel, selfId, playerRef]);

      return { suppressNextEvent };
    }
    ```

    Notes:
    - The hook does NOT subscribe a second time; it attaches handlers to the SAME channel that RoomClient created. supabase-js's `.on()` chains are idempotent only across distinct event subscriptions; the same `event` handler will not be attached twice in a single mount because the effect dep array is stable across renders.
    - Heartbeat fires for the host too. The host's own heartbeat handler is gated by `if (!isHost) return` on receipt — but since `broadcast.self: false`, the host doesn't receive its own heartbeat anyway. Belt-and-suspenders for safety.
    - `suppressNextEvent` is returned so the host's `onStateChange` in RoomClient can flip it before calling `seekTo` programmatically.

    import { HostBadge } from './HostBadge';

    type Props = {
      participants: readonly Participant[];
      host: Participant | null;
    };

    /**
     * PresenceList: chip-row strip + expanded panel.
     * D-322: always-visible chip-row. Mobile = horizontal scroll if overflow; desktop fits inline.
     * D-323: each row shows `{name} · {city}` plus the HostBadge on the elected host's row.
     *
     * The list aesthetic borrows from components/landing/CityPreview.tsx (numbered rows, mono

      // Sort by joined_at asc so host appears first (matches electHost ordering)

                  (isHostRow(p)
                    ? 'border-[color:var(--accent)]'
                    : 'border-[color:var(--line)]')

    'use client';

    import { useState, useTransition } from 'react';
    import type { RealtimeChannel } from '@supabase/supabase-js';

    import { CURATED_VIDEOS } from '@/lib/videos';
    import { pickVideo } from '@/app/actions/pick-video';

    type Props = {
      open: boolean;
      onClose: () => void;
      roomId: string;
      channel: RealtimeChannel | null;
      onPicked: (videoId: string) => void;
    };

    /**
     * VideoPickerSheet: host-only modal/sheet for choosing a video from CURATED_VIDEOS.
     * Selection flow (D-316):
     *   1. Call pickVideo(roomId, videoId) server action — writes rooms.youtube_video_id.
     *   2. On success, call onPicked(videoId) so RoomClient updates local videoId state
     *      (player re-renders with new prop; react-youtube tears down + recreates iframe).
     *   3. Broadcast sync_play{ timestamp: 0, videoId } so non-hosts swap to the new video
     *      and start at t=0. Note: SyncPlay's `videoId` is optional (room-types.ts);
     *      including it tells receivers to swap before seeking (RESEARCH.md Pattern 2).
     *
     * Layout: vanilla CSS sheet (no framer-motion / no radix). Mobile = slide-up bottom
     * sheet; desktop ≥md = centered modal. Click the backdrop to dismiss.
     */
    export function VideoPickerSheet({
      open,
      onClose,
      roomId,
      channel,
      onPicked,
    }: Props) {
      const [error, setError] = useState<string | null>(null);
      const [pending, startTransition] = useTransition();

      if (!open) return null;

      const onPick = (videoId: string) => {
        setError(null);
        startTransition(async () => {
          const result = await pickVideo(roomId, videoId);
          if ('error' in result) {
            setError(result.error);
            return;
          }
          onPicked(videoId);
          // Broadcast to peers. Optimistic — non-hosts swap + seek to 0.
          channel?.send({
            type: 'broadcast',
            event: 'sync_play',
            payload: { timestamp: 0, videoId },
          });
          onClose();
        });
      };

      return (
        <>
          {/* Backdrop */}
          <button
            type="button"
            aria-label="Close picker"
            onClick={onClose}

    import { useEffect, useRef, useState } from 'react';
    import type { RealtimeChannel } from '@supabase/supabase-js';

    import { createClient } from '@/lib/supabase/client';
    import { Player } from '@/components/room/Player';
    import { RoomHeader } from '@/components/room/RoomHeader';
    import { PresenceList } from '@/components/room/PresenceList';
    import { VideoPickerSheet } from '@/components/room/VideoPickerSheet';
    import { usePresence } from '@/hooks/usePresence';
    import { useRoomSync } from '@/hooks/useRoomSync';

    type Props = {
      signup: { id: string; name: string; city: string };
      room: {
        id: string;
        type: string;
        city: string;
        youtubeVideoId: string | null;
      };
    };

    export default function RoomClient({ signup, room }: Props) {
      const [channel, setChannel] = useState<RealtimeChannel | null>(null);
      const [videoId, setVideoId] = useState<string | null>(room.youtubeVideoId);
      const [pickerOpen, setPickerOpen] = useState(false);
      const playerRef = useRef<YouTubePlayer | null>(null);

      // Channel lifecycle (preserved verbatim from Plan 04)
      useEffect(() => {
        const supabase = createClient();
        const ch = supabase.channel(`room:${room.id}`, {
          config: {
            presence: { key: signup.id },
            broadcast: { self: false },
          },
        });

        ch.subscribe(async (status) => {
          if (status === 'SUBSCRIBED') {
            await ch.track({
              user_id: signup.id,
              name: signup.name,
              city: signup.city,
              joined_at: Date.now(),
            });
          }
        });

        setChannel(ch);

        return () => {
          ch.unsubscribe();
          supabase.removeChannel(ch);
          setChannel(null);
        };
      }, [room.id, signup.id, signup.name, signup.city]);

      const { participants, host, isHost, count } = usePresence(channel, signup.id);
      const { suppressNextEvent } = useRoomSync(channel, signup.id, isHost, playerRef);

      // Auto-open picker for host when room has no video (D-317)
      useEffect(() => {
        if (isHost && !videoId) setPickerOpen(true);
      }, [isHost, videoId]);

      // Host's onStateChange: broadcast sync_play / sync_pause / sync_seek
      const onStateChange = (e: YouTubeEvent<number>) => {
        if (!isHost) return;
        if (suppressNextEvent.current) {
          suppressNextEvent.current = false;
          return;
        }
        const ts = playerRef.current?.getCurrentTime() ?? 0;
        if (e.data === 1) {
          // PLAYING
          channel?.send({
            type: 'broadcast',
            event: 'sync_play',
            payload: { timestamp: ts },
          });
        } else if (e.data === 2) {
          // PAUSED
          channel?.send({
            type: 'broadcast',
            event: 'sync_pause',
            payload: { timestamp: ts },
          });
        }

      // Player error handler — embed disabled (101/150)
      const [playerError, setPlayerError] = useState<string | null>(null);
      const onError = (e: YouTubeEvent<number>) => {
        if (e.data === 101 || e.data === 150) {
          setPlayerError(
            isHost
              ? "This video can't be played here. Pick another."
              : 'Waiting for host to pick another video.',
          );
        }
      };

      // Capture player ref via Player.onReady
      const onReady = (player: YouTubePlayer) => {
        playerRef.current = player;
      };

      const hostName = host?.name ?? null;

      return (
        <main className="min-h-screen relative overflow-hidden flex flex-col bg-[color:var(--bg)]">
          <RoomHeader city={room.city} count={count} />

          <section className="flex-1 mx-auto max-w-6xl w-full px-6 py-8 flex flex-col gap-6">
            <div className="rise" style={{ animationDelay: '120ms' }}>
              <Player
                videoId={videoId}
                isHost={isHost}
                hostName={hostName}
                onReady={onReady}
                onStateChange={onStateChange}
                onError={onError}
              />
            </div>

            {playerError && (
              <p
                role="alert"
                className="font-mono text-[0.78rem] tracking-[0.04em] text-[color:#ff7878] border-l-2 border-[#ff7878] pl-3"
              >
                {playerError}
              </p>
            )}

            {isHost && (
              <button
                type="button"
                onClick={() => setPickerOpen(true)}
                className="cta self-start"
              >
                {videoId ? 'Change video' : 'Pick a video'}
                <span className="arrow" aria-hidden />
              </button>
            )}

            <PresenceList participants={participants} host={host} />
          </section>

          <VideoPickerSheet
            open={pickerOpen}
            onClose={() => setPickerOpen(false)}
            roomId={room.id}
            channel={channel}
            onPicked={(id) => setVideoId(id)}
          />
        </main>
      );
    }
