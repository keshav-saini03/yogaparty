    ```ts
    'use server';

    import { createAdminClient } from '@/lib/supabase/admin';
    import { CURATED_VIDEOS } from '@/lib/videos';

    const UUID_RE =
      /^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i;

    /**
     * Host-only write path to rooms.youtube_video_id. RLS posture matches Phase 2:
     * service-role admin client (anon would be blocked by 42501). Curated list is the
     * only valid videoId surface (D-010 banned free-form URLs; D-014 banned admin panel).
     *
     * Threats mitigated:
     * - T-03-02 (V5 input validation): videoId allow-listed against CURATED_VIDEOS.
     * - V5 input validation: roomId UUID-regex guarded.
     */
    export async function pickVideo(
      roomId: string,
      videoId: string,
    ): Promise<{ ok: true } | { error: string }> {
      if (!UUID_RE.test(roomId)) return { error: 'invalid room' };
      if (!CURATED_VIDEOS.find((v) => v.id === videoId)) {
        return { error: 'invalid video' };
      }

      const supabase = createAdminClient();
      const { error } = await supabase
        .from('rooms')
        .update({ youtube_video_id: videoId })
        .eq('id', roomId);

      if (error) {
        console.error('pickVideo update failed', error);
        return { error: 'update failed' };
      }
      return { ok: true };
    }
    ```

    Note: this file uses `'use server'` (NOT `import 'server-only'`) because it's a server action — Next.js requires that directive for the export to be callable from client components.

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Browser to signup action | Untrusted form data crosses; `name`, `phone`, `referrer_id`, `next` are all attacker-controllable. |
| Browser to pickVideo action | Untrusted args crosses; `roomId` and `videoId` are attacker-controllable (a host can claim to be host even if not — but the only damage is updating their own room's video; non-hosts in the same room don't expose anything new). |
| Signup action to Supabase | Service-role admin client; bypasses RLS. Already-trusted boundary inherited from Phase 2. |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-03-01 | Tampering | `?next=` open redirect | mitigate | NEXT_RE allow-list at server action: only `^/room/[0-9a-f-]{36}$` paths pass; everything else silently becomes `null` and the default city-room redirect fires. Implemented in Edit 2 of Task 1. |
| T-03-02 | Tampering | non-curated videoId in pickVideo | mitigate | `CURATED_VIDEOS.find(v => v.id === videoId)` allow-list check returns `{ error: 'invalid video' }` before any DB write. Implemented in Task 3. |
| T-03-03 | Spoofing | malformed roomId in pickVideo | mitigate | UUID_RE regex check; structured error before DB call. |
| T-03-04 | Information Disclosure | service-role key reach | mitigate | `pickVideo` is a `'use server'` module; `lib/rooms.ts` is `import 'server-only'`. Neither reaches the browser bundle. |
| T-03-05 | Repudiation | bad-actor sets cookie via XSS | accept | Cookie is `httpOnly: true`; React's default JSX escaping handles all user strings. No `dangerouslySetInnerHTML` introduced in this plan. |
</threat_model>
