    ```tsx
    'use client';

    import { useEffect, useState, useRef, type FormEvent } from 'react';
    import type { RealtimeChannel } from '@supabase/supabase-js';

    import { ChatMessage } from './ChatMessage';
    import type { ChatMsg } from '@/lib/room-types';

    type Props = {
      channel: RealtimeChannel | null;
      signup: { id: string; name: string };
    };

    /**
     * Chat: bottom sheet on mobile, right-side sidebar on desktop (>= md = 768px).
     * Ephemeral state per D-319 — local useState only; refresh clears history.
     *
     * Critical optimistic-append (Pitfall 9 / RESEARCH.md §1):
     *   broadcast.self: false means the SENDER does not receive their own broadcast.
     *   We MUST locally append the message to state on send, otherwise the sender
     *   sees nothing happen.
     *
     * Receiver: appends in receipt order (Realtime delivers FIFO per sender; cross-sender
     * order is not guaranteed but human-pace messaging makes that fine — D-320).
     */
    export function Chat({ channel, signup }: Props) {
      const [messages, setMessages] = useState<ChatMsg[]>([]);
      const [text, setText] = useState('');
      const [open, setOpen] = useState(false);
      const listRef = useRef<HTMLDivElement | null>(null);

      // Subscribe to incoming chat events
      useEffect(() => {
        if (!channel) return;
        channel.on('broadcast', { event: 'chat' }, ({ payload }) => {
          setMessages((prev) => [...prev, payload as ChatMsg]);
        });
        // No per-handler cleanup; channel teardown in RoomClient handles it.
      }, [channel]);

      // Auto-scroll to latest on new message
      useEffect(() => {
        const el = listRef.current;
        if (!el) return;
        el.scrollTop = el.scrollHeight;
      }, [messages.length]);

      const onSubmit = (e: FormEvent<HTMLFormElement>) => {
        e.preventDefault();
        const trimmed = text.trim();
        if (!trimmed || !channel) return;

        const msg: ChatMsg = {
          type: 'chat',
          user: signup.name,
          text: trimmed,
          timestamp: Date.now(),
        };

        // Optimistic local append (sender does NOT receive own broadcast)
        setMessages((prev) => [...prev, msg]);

        channel.send({
          type: 'broadcast',
          event: 'chat',
          payload: msg,
        });

        setText('');
      };

      return (
        <>
          {/* Mobile toggle handle — visible only below md */}
          <button
            type="button"
            onClick={() => setOpen((v) => !v)}
            className="md:hidden fixed bottom-0 left-0 right-0 z-30 border-t border-[color:var(--line)] bg-[color:var(--bg-raised)] py-3 font-mono text-[0.7rem] tracking-[0.2em] uppercase text-[color:var(--ink-mute)]"
            aria-controls="room-chat"
            aria-expanded={open}
          >
            {open ? '— Hide chat' : `Chat · ${messages.length}`}
          </button>

          <aside
            id="room-chat"
            aria-label="Room chat"
            className={
              // Mobile: fixed bottom sheet, transform open/close
              'fixed inset-x-0 bottom-0 z-40 max-h-[60vh] transform transition-transform duration-300 ' +
              (open ? 'translate-y-0' : 'translate-y-full') + ' ' +
              'border-t border-[color:var(--line)] bg-[color:var(--bg-raised)] flex flex-col ' +
              // Desktop: anchored right sidebar, always visible
              'md:static md:inset-auto md:max-h-none md:h-full md:w-80 md:translate-y-0 md:border-l md:border-t-0'
            }
          >
            <header className="px-4 py-3 border-b border-[color:var(--line)] flex items-baseline justify-between">
              <span className="eyebrow">Chat</span>
              <span className="font-mono text-[0.65rem] tracking-[0.2em] uppercase text-[color:var(--ink-mute)]">
                Ephemeral
              </span>
            </header>

            <div
              ref={listRef}
              className="flex-1 overflow-y-auto px-4 py-2 min-h-0"
            >
              {messages.length === 0 && (
                <p className="font-mono text-[0.72rem] tracking-[0.16em] uppercase text-[color:var(--ink-mute)]">
                  Say hi to start the room.
                </p>
              )}
              {messages.map((m, i) => (
                <ChatMessage key={`${m.timestamp}-${i}`} user={m.user} text={m.text} />
              ))}
            </div>

            <form
              onSubmit={onSubmit}
              className="px-4 py-3 border-t border-[color:var(--line)] flex gap-2"
            >
              <input
                type="text"
                value={text}
                onChange={(e) => setText(e.target.value)}
                placeholder="Message"
                aria-label="Send a message"
                maxLength={500}
                className="field flex-1"
              />
              <button
                type="submit"
                disabled={!text.trim() || !channel}
                className="cta shrink-0"
              >
                Send
              </button>
            </form>
          </aside>
        </>
      );
    }
    ```

    Notes:
    - `maxLength={500}` is a soft client-side guard. No server validation because there's no server in the loop (broadcast-only).
    - Auto-scroll uses `scrollTop = scrollHeight` after every message; replace with smarter behavior (e.g., only scroll if user was already at bottom) is a Phase 7 polish concern.
    - The mobile toggle button counts unread messages even when open — acceptable for v1; UX improvement is out of scope.

    ```tsx
    import { describe, it, expect, vi } from 'vitest';
    import { render, screen, fireEvent } from '@testing-library/react';

    import { Chat } from './Chat';
    import type { ChatMsg } from '@/lib/room-types';

    // Minimal channel mock matching the surface Chat.tsx uses:
    //   channel.on('broadcast', { event: 'chat' }, handler)
    //   channel.send({ type: 'broadcast', event: 'chat', payload })
    function makeChannel() {
      const handlers: Array<(arg: { payload: ChatMsg }) => void> = [];
      const send = vi.fn();
      const on = vi.fn(
        (
          _kind: 'broadcast',
          filter: { event: string },
          handler: (arg: { payload: ChatMsg }) => void,
        ) => {
          if (filter.event === 'chat') handlers.push(handler);
          return channel; // chainable
        },
      );
      const channel = { on, send } as unknown as Parameters<typeof Chat>[0]['channel'];
      const emit = (msg: ChatMsg) => handlers.forEach((h) => h({ payload: msg }));
      return { channel, send, emit };
    }

    const sig = { id: 'u1', name: 'ALICE' };

    function asMsg(user: string, text: string, ts: number): ChatMsg {
      return { type: 'chat', user, text, timestamp: ts };
    }

    describe('Chat', () => {
      it('renders incoming messages in receipt order', () => {
        const { channel, emit } = makeChannel();
        render(<Chat channel={channel} signup={sig} />);

        emit(asMsg('BOB', 'first', 1));
        emit(asMsg('CARA', 'second', 2));
        emit(asMsg('DOM', 'third', 3));
        emit(asMsg('EVE', 'fourth', 4));
        emit(asMsg('FRAN', 'fifth', 5));

        const texts = screen.getAllByText(/first|second|third|fourth|fifth/);
        const order = texts.map((el) => el.textContent);
        expect(order).toEqual(['first', 'second', 'third', 'fourth', 'fifth']);
      });

      it('optimistically appends sent message to the local list', () => {
        const { channel, send } = makeChannel();
        render(<Chat channel={channel} signup={sig} />);

        const input = screen.getByLabelText('Send a message') as HTMLInputElement;
        fireEvent.change(input, { target: { value: 'hello' } });

        // Submit via the form (Send button)
        const form = input.closest('form')!;
        fireEvent.submit(form);

        expect(send).toHaveBeenCalledWith(
          expect.objectContaining({
            type: 'broadcast',
            event: 'chat',
            payload: expect.objectContaining({ user: 'ALICE', text: 'hello' }),
          }),
        );
        // And the message is visible in the rendered list (optimistic — broadcast.self false)
        expect(screen.getByText('hello')).toBeInTheDocument();
      });
    });
    ```

    - `asMsg` includes `type: 'chat'` because the ChatMsg union member requires it.
    - The optimistic test uses `closest('form')` because there's no submit button accessible name configured uniquely.
