    Create /Users/habuild/Desktop/work/Habuild/hack-a-thon/lib/sync-utils.ts with EXACTLY this body:

    ```ts
    // Pure utility functions for room sync. Imported by hooks/usePresence.ts and hooks/useRoomSync.ts.
    // Kept pure (no React, no Supabase) so they can be unit-tested without mocks.
    // See .planning/phases/03-city-watch-room/03-RESEARCH.md §3 and §4 for derivation.

    type Participant = {
      user_id: string;
      name: string;
      city: string;
      joined_at: number;
    };

    /**
     * Deterministic host election: earliest `joined_at` wins; tiebreak by `user_id` lex sort.
     * Every client computes this independently; result must be identical across clients
     * because `joined_at` is shared via presence state. (PROJECT.md D-008, CONTEXT.md D-309.)
     */
    export function electHost(
      participants: readonly Participant[],
    ): Participant | null {
      if (participants.length === 0) return null;
      return [...participants].sort(
        (a, b) =>
          a.joined_at - b.joined_at || a.user_id.localeCompare(b.user_id),
      )[0];
    }

    /**
     * Drift detector. Returns true iff |hostTime - clientTime| strictly exceeds threshold.
     * Default threshold is 2 seconds per PROJECT.md C-008. Strict `>` (NOT `>=`) — exactly
     * 2.000 seconds is in spec; only 2.001+ triggers correction.
     */
    export function shouldCorrect(
      hostTime: number,
      clientTime: number,
      threshold = 2,
    ): boolean {
      return Math.abs(hostTime - clientTime) > threshold;
    }
    ```

    import { describe, it, expect } from 'vitest';
    import { electHost, shouldCorrect } from '@/lib/sync-utils';

    type P = { user_id: string; name: string; city: string; joined_at: number };
    const make = (id: string, joined: number): P =>
      ({ user_id: id, name: id, city: 'X', joined_at: joined });

    describe('electHost', () => {
      it('returns null for empty list', () => {
        expect(electHost([])).toBeNull();
      });
      it('returns single participant when only one', () => {
        const p = make('a', 100);
        expect(electHost([p])).toEqual(p);
      });
      it('earliest joined_at wins', () => {
        const a = make('a', 100);
        const b = make('b', 50);
        expect(electHost([a, b])).toEqual(b);
      });
      it('ties broken by user_id lex sort', () => {
        const a = make('a', 100);
        const b = make('b', 100);
        expect(electHost([b, a])).toEqual(a);
      });
    });

    describe('shouldCorrect', () => {
      it('returns false at exactly 2.0s threshold', () => {
        expect(shouldCorrect(10, 12)).toBe(false);
        expect(shouldCorrect(12, 10)).toBe(false);
      });
      it('returns true above 2.0s', () => {
        expect(shouldCorrect(10, 12.001)).toBe(true);
        expect(shouldCorrect(10, 7.999)).toBe(true);
      });
      it('respects custom threshold', () => {
        expect(shouldCorrect(10, 12, 5)).toBe(false);
        expect(shouldCorrect(10, 16, 5)).toBe(true);
      });
    });
