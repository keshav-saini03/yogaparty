    ```ts
    import 'server-only';

    import { createAdminClient } from '@/lib/supabase/admin';

    /**
     * Find-or-create the active `type='city'` room for the given city.
     *
     * Behavior locked by D-302 (city-room resolution shared between signup paths) +
     * D-317 (new rooms have youtube_video_id = NULL — host picks first).
     *
     * Schema preserved per D-309: no host_id column, youtube_video_id stays nullable,
     * no UNIQUE constraint added — race is handled at the application level via
     * a follow-up SELECT (Option B in 03-RESEARCH.md §6).
     *
     * City normalization: trim whitespace; preserve case (Mumbai, not mumbai — header
     * displays the value verbatim per RESEARCH.md §6). Null/empty falls back to a single
     * 'GLOBAL' sentinel room (acceptable hackathon scope; vanishingly rare on Vercel).
     */
    export async function findOrCreateCityRoom(
      rawCity: string | null,
    ): Promise<{ id: string }> {
      const cityKey = (rawCity ?? 'GLOBAL').trim() || 'GLOBAL';
      const supabase = createAdminClient();

      // 1. Find existing active city room
      const { data: existing } = await supabase
        .from('rooms')
        .select('id')
        .eq('type', 'city')
        .eq('city', cityKey)
        .eq('is_active', true)
        .maybeSingle();
      if (existing) return { id: existing.id };

      // 2. Create — youtube_video_id intentionally NULL (host picks per D-317)
      const { data: created, error } = await supabase
        .from('rooms')
        .insert({ type: 'city', city: cityKey, is_active: true })
        .select('id')
        .single();
      if (created) return { id: created.id };

      // 3. Race-loser: another caller created concurrently. Re-read.
      if (error) {
        const { data: retry } = await supabase
          .from('rooms')
          .select('id')
          .eq('type', 'city')
          .eq('city', cityKey)
          .eq('is_active', true)
          .maybeSingle();
        if (retry) return { id: retry.id };
      }

      console.error('findOrCreateCityRoom failed', { cityKey, error });
      throw new Error(`findOrCreateCityRoom failed for city=${cityKey}`);
    }
    ```
